cmake_minimum_required(VERSION 3.16)
project(SuperJupiterTimeAggregateConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_TIME_AGGREGATE_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_TIME_AGGREGATE_SOURCE_DIR}" time-aggregate)
  if(TARGET superjupiter-time-aggregate-tests OR TARGET superjupiter-time-aggregate-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterTimeAggregate 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-time-aggregate-consumer main.cpp)
target_link_libraries(superjupiter-time-aggregate-consumer PRIVATE SuperJupiter::TimeAggregate)
